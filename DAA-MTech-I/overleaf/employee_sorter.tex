\section{Problem 1: Employee Name Sorting and Duplicate Detection}

\subsection{Problem Overview and Objectives}
An organization maintains an unsorted list of $N$ employee names. The objective is to design an efficient algorithm that:
\begin{enumerate}
    \item Arranges the employee names in strict lexicographical order \textbf{without using built-in sorting functions} (such as Python's \texttt{sort()} or \texttt{sorted()}).
    \item Identifies and displays all duplicate employee names along with their exact occurrence counts. If no duplicates exist, the algorithm reports the same.
    \item Evaluates non-sorting alternative strategies for duplicate detection (e.g., Hash Tables, Tries, Brute Force).
\end{enumerate}

\subsection{Specifications}
\begin{itemize}
    \item \textbf{Input}: An unsorted array/list \texttt{employee\_names} containing $N$ string elements, where each string has a maximum length of $K$ characters.
    \item \textbf{Output}: 
    \begin{itemize}
        \item \texttt{sorted\_employee\_names}: A list containing all $N$ names arranged in lexicographical order.
        \item \texttt{duplicate\_names\_map}: A dictionary/map recording names with occurrence frequency $> 1$.
    \end{itemize}
    \item \textbf{Constraints}: Standard high-level sorting library primitives are strictly prohibited. Explicit character-by-character comparison must be performed.
\end{itemize}

\subsection{Character Case Treatment and String Comparison}
In languages like C, string variables are character pointers (\texttt{char*}), and relational operators (\texttt{<}, \texttt{>}, \texttt{==}) compare memory addresses rather than string content. To maintain language-agnostic precision, string content comparison is implemented via an explicit character-by-character comparison function, \texttt{compare\_strings}, equivalent to C's \texttt{strcmp()}.

\begin{itemize}
    \item \textbf{ASCII Precedence}: Character comparison follows standard ASCII ordinal values:
    \[
    \text{'A'} (65) < \text{'B'} (66) < \dots < \text{'Z'} (90) < \text{'a'} (97) < \text{'b'} (98) < \dots < \text{'z'} (122)
    \]
    \item \textbf{Duplicate Equality}: Two names $s_1$ and $s_2$ are duplicates if and only if they are identical in both content and case ($s_1 == s_2$). For example, \texttt{"Alice"} and \texttt{"alice"} are distinct names, whereas two occurrences of \texttt{"John"} constitute a duplicate pair.
\end{itemize}

\subsection{Formal Algorithm Design}

\begin{algorithm}[H]
\caption{Explicit Character-by-Character String Comparison}
\begin{algorithmic}[1]
\Function{CompareStrings}{$s_1, s_2$}
    \State $L_1 \gets \text{Length}(s_1)$, $L_2 \gets \text{Length}(s_2)$
    \State $M \gets \min(L_1, L_2)$
    \For{$i \gets 0$ \textbf{to} $M - 1$}
        \If{$s_1[i] \neq s_2[i]$}
            \If{$s_1[i] < s_2[i]$} \Return $-1$
            \Else \ \Return $1$
            \EndIf
        \EndIf
    \EndFor
    \If{$L_1 < L_2$} \Return $-1$
    \ElsIf{$L_1 > L_2$} \Return $1$
    \Else \ \Return $0$
    \EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Custom Merge Sort and Duplicate Detection}
\begin{algorithmic}[1]
\Function{CustomMerge}{$left, right$}
    \State $merged \gets []$, $i \gets 0$, $j \gets 0$
    \While{$i < \text{Length}(left)$ \textbf{and} $j < \text{Length}(right)$}
        \If{\Call{CompareStrings}{$left[i], right[j]$} $\le 0$}
            \State Append $left[i]$ to $merged$; $i \gets i + 1$
        \Else
            \State Append $right[j]$ to $merged$; $j \gets j + 1$
        \EndIf
    \EndWhile
    \State Append remaining elements of $left$ and $right$ to $merged$
    \State \Return $merged$
\EndFunction
\Statex
\Function{MergeSortNames}{$names$}
    \State $N \gets \text{Length}(names)$
    \If{$N \le 1$} \Return $names$ \EndIf
    \State $mid \gets \lfloor N / 2 \rfloor$
    \State $left\_sorted \gets$ \Call{MergeSortNames}{$names[0 \dots mid-1]$}
    \State $right\_sorted \gets$ \Call{MergeSortNames}{$names[mid \dots N-1]$}
    \State \Return \Call{CustomMerge}{$left\_sorted, right\_sorted$}
\EndFunction
\Statex
\Function{FindDuplicatesAfterSorting}{$sorted\_names$}
    \State $map \gets \{\}$, $N \gets \text{Length}(sorted\_names)$, $i \gets 0$
    \While{$i < N$}
        \State $count \gets 1$, $j \gets i + 1$
        \While{$j < N$ \textbf{and} \Call{CompareStrings}{$sorted\_names[i], sorted\_names[j]$} $== 0$}
            \State $count \gets count + 1$; $j \gets j + 1$
        \EndWhile
        \If{$count > 1$}
            \State $map[sorted\_names[i]] \gets count$
        \EndIf
        \State $i \gets j$
    \EndWhile
    \State \Return $map$
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Complexity Analysis and Mathematical Derivation}
Let $N$ be the total number of employee names and $K$ be the maximum length of an employee name string.

\subsubsection{Time Complexity Derivation}
\begin{enumerate}
    \item \textbf{Recursion Tree Depth}: Merge sort recursively partitions an array of size $N$ into two equal halves until subproblems reach size 1. The height of the recursion tree is $\log_2 N$.
    \item \textbf{Work Per Recursion Level}: At each level $d \in [0, \log_2 N]$, the algorithm merges sub-lists containing a total of $N$ elements. Comparing two string elements character-by-character requires at most $K$ character operations. Thus, total work per level is $O(N \cdot K)$.
    \item \textbf{Sorting Time}:
    \[
    T_{\text{sort}}(N) = \text{Work per level} \times \text{Number of levels} = O(N \cdot K \log N)
    \]
    \textbf{Note on Asymptotic Notation}: The complexity is strictly $O(N \cdot K \log N)$, \textbf{not} $O(N \log(N \cdot K))$, because the logarithmic factor arises solely from array partitioning ($N$), while string length $K$ acts as a linear multiplier per character comparison.
    \item \textbf{Linear Duplicate Scan Time}: Once sorted, identical names occupy adjacent positions. Scanning $N$ elements with adjacent $K$-character comparisons takes $T_{\text{scan}}(N) = O(N \cdot K)$ time.
    \item \textbf{Total Time Complexity}:
    \[
    T_{\text{total}}(N) = O(N \cdot K \log N) + O(N \cdot K) = O(N \cdot K \log N)
    \]
    This bound holds across \textbf{Best-Case}, \textbf{Average-Case}, and \textbf{Worst-Case} configurations.
\end{enumerate}

\subsubsection{Auxiliary Space Complexity}
\begin{itemize}
    \item Recursion stack depth requires $O(\log N)$ frame allocations.
    \item Merging arrays creates temporary sub-lists of aggregate size $O(N \cdot K)$.
    \item \textbf{Total Auxiliary Space}: $O(N \cdot K)$.
\end{itemize}

\textbf{Conclusion}: When both lexicographical ordering and duplicate detection are required, Merge Sort + Linear Scan is optimal because sorting is mandatory and scanning adds zero asymptotic overhead. If only duplicate detection is required, the Hash Table approach is asymptotically faster ($O(N K)$).
