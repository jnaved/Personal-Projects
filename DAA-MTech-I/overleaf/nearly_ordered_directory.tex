\section{Problem 5: Nearly Ordered Employee Directory}

\subsection{Problem Overview and Objectives}
An employee directory containing $n$ names is expected to be in lexicographical order, but due to a data-entry error, some names may have been placed incorrectly.

Given the list of names:
\begin{enumerate}
    \item Determine whether the list is already sorted.
    \item Identify all inversion indices where ordering is violated ($names[i] > names[i+1]$).
    \item Determine whether exchanging \textbf{ONLY TWO} names can restore complete lexicographical order. If possible, report the two positions $(pos_1, pos_2)$.
\end{enumerate}
\textbf{Constraint}: Do \textbf{NOT} sort the complete array to determine ordering.

\subsection{Specifications}
\begin{itemize}
    \item \textbf{Input}: \texttt{names}, a list of $n$ string names ($K = \text{max name length}$).
    \item \textbf{Output}: \texttt{(is\_sorted, violation\_indices, can\_fix\_with\_single\_swap, swap\_positions)}.
\end{itemize}

\subsection{Conditions Under Which a Single Swap Restores Order}
A single swap of elements at indices $pos_1$ and $pos_2$ ($pos_1 < pos_2$) can restore complete lexicographical order if and only if the original list satisfies the following inversion point properties:

\begin{enumerate}
    \item \textbf{Inversion Count Bound}: The list contains at most 2 inversion points (indices $i$ where $names[i] > names[i+1]$).
    \item \textbf{Adjacent Swap Condition (1 Inversion)}:
    If exactly 1 inversion occurs at index $i$, the violation is caused by adjacent elements being swapped. Candidate swap positions:
    \[
    pos_1 = i, \quad pos_2 = i + 1
    \]
    \item \textbf{Non-Adjacent Swap Condition (2 Inversions)}:
    If 2 inversions occur at indices $i$ and $j$ ($i < j$), the violation is caused by non-adjacent elements being swapped. Candidate swap positions:
    \[
    pos_1 = i, \quad pos_2 = j + 1
    \]
    \item \textbf{Multiple Violation Threshold ($> 2$ Inversions)}:
    If more than 2 inversions exist, exchanging a single pair cannot fix all inversion boundaries simultaneously.
\end{enumerate}

\subsection{Formal Algorithm Design}

\begin{algorithm}[H]
\caption{Nearly Ordered Directory Analysis and Single Swap Restoration}
\begin{algorithmic}[1]
\Function{AnalyzeNearlyOrderedDirectory}{$names$}
    \State $n \gets \text{Length}(names)$
    \If{$n \le 1$} \Return $(\text{True}, [], \text{True}, \text{None})$ \EndIf
    \State $violations \gets []$ \Comment{Phase 1: Inversion Scanning}
    \For{$i \gets 0$ \textbf{to} $n - 2$}
        \If{\Call{CompareStrings}{$names[i], names[i+1]$} $> 0$}
            \State Append $i$ to $violations$
        \EndIf
    \EndFor
    \If{$\text{Length}(violations) == 0$}
        \State \Return $(\text{True}, [], \text{True}, \text{None})$
    \EndIf
    \State $pos_1 \gets \text{None}, pos_2 \gets \text{None}$ \Comment{Phase 2: Candidate Identification}
    \If{$\text{Length}(violations) == 1$}
        \State $pos_1 \gets violations[0]$, $pos_2 \gets violations[0] + 1$
    \ElsIf{$\text{Length}(violations) == 2$}
        \State $pos_1 \gets violations[0]$, $pos_2 \gets violations[1] + 1$
    \Else
        \State \Return $(\text{False}, violations, \text{False}, \text{None})$ \Comment{$>2$ violations cannot be fixed by 1 swap}
    \EndIf
    \State $candidate \gets \text{Copy}(names)$ \Comment{Phase 3: Swap Verification}
    \State Swap $candidate[pos_1]$ and $candidate[pos_2]$
    \State $is\_restored \gets \text{True}$
    \For{$i \gets 0$ \textbf{to} $n - 2$}
        \If{\Call{CompareStrings}{$candidate[i], candidate[i+1]$} $> 0$}
            \State $is\_restored \gets \text{False}$; \textbf{break}
        \EndIf
    \EndFor
    \If{$is\_restored$}
        \State \Return $(\text{False}, violations, \text{True}, (pos_1, pos_2))$
    \Else
        \State \Return $(\text{False}, violations, \text{False}, \text{None})$
    \EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\subsection{Mathematical Complexity Analysis}
Let $n$ be the number of names and $K$ be the maximum string length.

\subsubsection{Time Complexity Derivation}
\begin{itemize}
    \item \textbf{Phase 1 (Inversion Scan)}: Scans $n-1$ adjacent pairs taking $(n-1) \times O(K) = O(n \cdot K)$ time.
    \item \textbf{Phase 2 (Candidate Selection)}: Evaluates list length in $O(1)$ time.
    \item \textbf{Phase 3 (Swap Verification Pass)}: Copies array in $O(n)$ time and verifies sorted order in $(n-1) \times O(K) = O(n \cdot K)$ time.
    \item \textbf{Total Time Complexity}:
    \[
    T_{\text{total}}(n) = O(n \cdot K) + O(1) + O(n \cdot K) = O(n \cdot K)
    \]
    This strictly obeys the constraint, avoiding full array sorting which requires $O(n \log n \cdot K)$ time.
\end{itemize}

\subsubsection{Auxiliary Space Complexity}
\begin{itemize}
    \item \textbf{Auxiliary Space Complexity}: $O(1)$ (or $O(n)$ if temporary array copy is created for verification; verification can be done in-place with scalar bounds checking in $O(1)$ space).
\end{itemize}
